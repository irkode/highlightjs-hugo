{{- $lang := . }} {{- $data := index hugo.Data.keywords $lang }} {{- $fixed := $data.fixed }} {{- if not $fixed }} {{- errorf "data/keywords/%s.yaml has no `fixed` section" $lang }} {{- end }} {{- /* temporary storage for words ordered by scope and length */ -}} {{- $pad := newScratch }} {{- /* seed the scopes with the fixed word lists */ -}} {{- $fixedWords := slice }} {{- range $scope, $words := $fixed }} {{- $fixedWords = $fixedWords | append $words }} {{- range $word := $words }} {{- $len := printf "%02d" (len .) }} {{- with $pad.Get $scope }} {{- with index . $len }} {{- $pad.SetInMap $scope $len (append $word .) }} {{- else }} {{- $pad.SetInMap $scope $len (slice $word) }} {{- end }} {{- else }} {{- $pad.SetInMap $scope $len (slice $word) }} {{- end }} {{- end }} {{- end }} {{- /* hugoDocs documents Hugo only -- `go` is entirely the fixed lists above */ -}} {{- $docsKeywords := slice }} {{- if eq $lang "hugo" }} {{- $functions := hugo.Data.hugodocs.functions }} {{- if not $functions }} {{- errorf "hugo.Data.hugodocs.functions is empty -- work/hugodocs/pinned.json is missing or unreadable; regenerate it with `.\\build.ps1 -Steps ExtractKeywords`" }} {{- end }} {{- range $name, $entry := $functions }} {{- /* a function and each of its aliases are classified the same way */ -}} {{- range $word := (append $name $entry.aliases) }} {{- if eq 0 (len $word) }} {{- continue }} {{- end }} {{- /* fixed keywords already added to their scope above */ -}} {{- if in $fixedWords $word }} {{- continue }} {{- end }} {{- $scope := "BUILTINS" }} {{- if eq $entry.section "go-template" }} {{- $scope = "KEYWORDS" }} {{- $docsKeywords = $docsKeywords | append $word }} {{- else if strings.Contains $word "." }} {{- $scope = "FUNCTIONS" }} {{- end }} {{- $len := printf "%02d" (len $word) }} {{- with $pad.Get $scope }} {{- with index . $len }} {{- $pad.SetInMap $scope $len (append $word .) }} {{- else }} {{- $pad.SetInMap $scope $len (slice $word) }} {{- end }} {{- else }} {{- $pad.SetInMap $scope $len (slice $word) }} {{- end }} {{- end }} {{- end }} {{- /* fail loudly: a go-template construct missing from `patterns` mis-highlights */ -}} {{- $patterns := $data.patterns }} {{- $patternWords := slice }} {{- range $name, $words := $patterns }} {{- range $w := $words }} {{- $plain := replace $w `\s+` " " }} {{- /* skip composites like "else if" -- their words are listed separately */ -}} {{- if not (strings.Contains $plain " ") }} {{- if not (in $patternWords $plain) }} {{- $patternWords = $patternWords | append $plain }} {{- end }} {{- end }} {{- end }} {{- end }} {{- $undeclared := slice }} {{- $stale := slice }} {{- range $w := $docsKeywords }} {{- if not (in $patternWords $w) }}{{ $undeclared = $undeclared | append $w }}{{ end }} {{- end }} {{- range $w := $patternWords }} {{- if not (in $docsKeywords $w) }}{{ $stale = $stale | append $w }}{{ end }} {{- end }} {{- if or $undeclared $stale }} {{- errorf (printf "hugoDocs go-template section no longer matches `patterns` in data/keywords/%s.yaml -- new in the docs: [%s]; in patterns but gone from the docs: [%s]. Update KEYWORDS_PIPELINE / KEYWORDS_STANDALONE (or add the word to `fixed.BUILTINS` if it should highlight as a builtin)." $lang (delimit (sort $undeclared) ", ") (delimit (sort $stale) ", ")) }} {{- end }} {{- end }} {{- /* flatten the scratch pad to a map{scope: [words]} NOTE: length keys are zero padded to ensure correct sort order, max length is 99 */ -}} {{- $keywords := dict }} {{- range $scope, $values := $pad.Values }} {{- $names := slice }} {{- /* collect all strings for the current scope, sort by length(desc) and name(asc)*/ -}} {{- range sort $values | collections.Reverse }} {{- $names = $names | append (sort .) }} {{- end }} {{- $keywords = merge $keywords (dict $scope $names) }} {{- end }} {{- return $keywords }} {{- /* remove all trailing whitespace */ -}}
